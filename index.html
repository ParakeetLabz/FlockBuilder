Sixty seconds to a game

<script type="module">
import { Game, Scene, Sprite } from './src/flock.js';

const game = new Game({ width: 800, height: 600 });
const scene = new Scene(game);
const ball = scene.add(new Sprite({
  x: 100, y: 100, w: 40, h: 40,
  shape: 'circle', color: '#22d3ee'
}));
ball.vel.set(240, 180);
scene.update = (dt) => {
  ball.pos.x += ball.vel.x * dt;
  ball.pos.y += ball.vel.y * dt;
  if (ball.x < 0 || ball.x + ball.w > 800) ball.vel.x *= -1;
  if (ball.y < 0 || ball.y + ball.h > 600) ball.vel.y *= -1;
};
game.pushScene(scene);
game.start();
</script>
